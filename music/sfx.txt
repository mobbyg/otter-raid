# Otter Raid sound effects (all play on SID voice 3).
# See docs/c64-sound.md.  Numbers come from c64/sound/sound_ids.asm.
#
# sfx NAME prio N        higher priority wins, equal restarts
#   setup A D S R PW PWSPD CTRL   envelope, pulse width, pulse change
#                        per frame and control in one command
#   adsr A D S R         envelope (hex nibbles)
#   ctrl XX              control register incl. gate bit (41 = pulse
#                        on, 40 = pulse released, 81 noise, 11 tri...)
#   pw $XXX              pulse width      pwspd N  pulse change/frame
#   note NOTE [WAIT]     pitch from the note table, optional wait
#                        (a ctrl right before it is merged into it)
#   freq $XXXX           raw frequency    slide N  raw change/frame
#   sweep N1 N2 FRAMES [keep]   glide N1 -> N2, waits FRAMES; the
#                        slide stops afterwards unless 'keep'
#   wait N               frames
#   loop N ... next      repeat a block (must contain a wait)
# end
#
# Every effect starts after a one-frame hard restart of voice 3.
# Keep the commands between two waits few: they all run in one
# frame (cycle budget, see docs/c64-sound.md).

# ----- menus -----

sfx MENU_MOVE prio 1
  setup 0 0 a 4 $600 0 41
  note e6 2
  note b6 2
  ctrl 40
  wait 5
end

sfx MENU_SELECT prio 3
  setup 0 0 c 8 $800 64 41
  note c6 2
  note e6 2
  note g6 2
  note c7 3
  note g6 2
  note c7 6
  ctrl 40
  wait 14
end

# ----- danger -----

sfx GATOR_CHOMP prio 6
  setup 0 0 c 2 $400 0 81
  loop 2
    ctrl 81
    note a#7 1
    note a6 1
    ctrl 21
    sweep a3 c#2 5
    ctrl 20
    wait 3
  next
  setup 0 8 0 0 $400 0 81
  sweep e6 f5 8
  ctrl 80
  wait 2
end
